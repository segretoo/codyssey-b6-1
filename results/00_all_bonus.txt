[B1-d] NOT IN의 함정: 서브쿼리 결과에 NULL이 하나라도 있으면 결과가 0건이 된다
======================================================================

SELECT 'NOT IN (1, NULL)' AS expr, COUNT(*) AS rows_returned FROM (SELECT 1 WHERE 5 NOT IN (1, NULL))
UNION ALL
SELECT 'NOT IN (1, 2)',            COUNT(*)                   FROM (SELECT 1 WHERE 5 NOT IN (1, 2));

[B3-1] 지표① 착용당 비용(가격 ÷ 착용 횟수)이 가장 높은 옷 TOP 5 — "비싼데 안 입는 옷"
======================================================================
-- [SQLite 전용] 정수 나눗셈 방지를 위해 * 1.0 사용
SELECT c.name AS clothing_name,
       c.price,
       COUNT(*) AS wear_count,
       CAST(ROUND(c.price * 1.0 / COUNT(*)) AS INTEGER) AS cost_per_wear
FROM clothing c
INNER JOIN wear_log w ON w.clothing_id = c.clothing_id
WHERE c.price IS NOT NULL
GROUP BY c.clothing_id, c.name, c.price
ORDER BY cost_per_wear DESC
LIMIT 5;

[B3-2] 지표② 월별 착용 횟수 추이 (10월은 14일까지의 부분 집계)
======================================================================
-- [SQLite 전용] strftime('%Y-%m', ...)  ※ MySQL: DATE_FORMAT(worn_on, '%Y-%m')
SELECT strftime('%Y-%m', worn_on) AS month,
       COUNT(*)                   AS wear_count
FROM wear_log

ORDER BY month;

[B3-3] 지표③ 회원별 옷장 활용률 = 최근 30일 안에 입은 옷 수 ÷ 보유 옷 수
======================================================================
-- [SQLite 전용] date()
SELECT m.nickname,
       COUNT(DISTINCT c.clothing_id) AS owned,
       COUNT(DISTINCT CASE WHEN w.worn_on >= date('2026-10-14', '-30 days') THEN c.clothing_id END) AS worn_recently,
       ROUND(100.0 * COUNT(DISTINCT CASE WHEN w.worn_on >= date('2026-10-14', '-30 days') THEN c.clothing_id END)
                   / COUNT(DISTINCT c.clothing_id), 1) AS utilization_pct
FROM member m
INNER JOIN clothing c ON c.member_id = m.member_id AND c.status = '보유'
LEFT JOIN  wear_log w ON w.clothing_id = c.clothing_id
GROUP BY m.member_id, m.nickname
ORDER BY utilization_pct DESC, m.member_id;
