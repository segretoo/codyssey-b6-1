[B1-a] LEFT JOIN + IS NULL 방식
======================================================================
SELECT c.clothing_id, c.name
FROM clothing c
LEFT JOIN wear_log w ON w.clothing_id = c.clothing_id
WHERE w.wear_id IS NULL AND c.status = '보유'
ORDER BY c.clothing_id;
