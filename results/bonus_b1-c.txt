[B1-c] NOT IN 서브쿼리 방식
======================================================================
SELECT c.clothing_id, c.name
FROM clothing c
WHERE c.status = '보유'
  AND c.clothing_id NOT IN (SELECT clothing_id FROM wear_log)
ORDER BY c.clothing_id;
