[B1-b] NOT EXISTS 서브쿼리 방식
======================================================================
SELECT c.clothing_id, c.name
FROM clothing c
WHERE c.status = '보유'
  AND NOT EXISTS (SELECT 1 FROM wear_log w WHERE w.clothing_id = c.clothing_id)
ORDER BY c.clothing_id;
